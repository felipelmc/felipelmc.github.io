% Formatação do currículo. Nada aqui precisa ser tocado no dia a dia: para
% editar o conteúdo, mexa apenas em cv-en.qmd e cv-pt.qmd.
% ---------------------------------------------------------------------------

\usepackage{fontawesome5}
\usepackage{titlesec}
\usepackage{enumitem}
\usepackage{xcolor}
\usepackage{array}
\usepackage{needspace}
\usepackage{etoolbox}
\usepackage{fancyhdr}
\usepackage{lastpage}

% Justificação com protrusão e expansão. Sem isso, as linhas longas e
% justificadas do corpo abrem buracos visíveis. O Quarto já carrega o
% microtype sem opções, então aqui só se ajusta a configuração.
\usepackage{microtype}
\microtypesetup{protrusion=true, expansion=true, tracking=true}
\SetTracking{encoding=*, shape=sc}{40}   % versalete pede entreletra

\definecolor{cvmuted}{gray}{0.42}
\definecolor{cvrule}{gray}{0.60}
\definecolor{cvlink}{HTML}{7A2E2E}       % vinho; o azul puro pinta demais a página

\setlength{\parindent}{0pt}
\setlength{\parskip}{0.062cm}

% --- Nome + linha de contato ------------------------------------------------

\newcommand{\cvname}[1]{%
  \begingroup
  \centering
  {\huge\scshape #1\par}%
  \nointerlineskip
  \vspace{5pt}%
  {\color{cvrule}\hrule height 0.6pt}%
  \endgroup
  \vspace{9pt}%
}

% O título do YAML (title:) vira o cabeçalho do currículo.
\makeatletter
\renewcommand{\maketitle}{\cvname{\@title}}
\makeatother

% Linha de contato: o bloco ::: contact do .qmd cai aqui.
\newenvironment{cvcontact}
  {\par\centering\small}
  {\par\vspace{0.22cm}}

\newcommand{\cvsep}{\hspace{0.45em}\textperiodcentered\hspace{0.45em}}

% --- Seções (\section) ------------------------------------------------------

\titleformat{\section}
  [hang]
  {\normalfont\large\scshape}
  {}{0pt}
  {}
  [\vspace{-0.30cm}{\color{cvrule}\rule{\textwidth}{0.4pt}}]
\titlespacing*{\section}{0pt}{0.36cm}{0.10cm}

% --- Subseções (\subsection): agrupamentos dentro de uma seção --------------
% Versalete no corpo do texto: separa da entrada por forma, não por tamanho.

\titleformat{\subsection}
  [hang]
  {\normalfont\normalsize\scshape}
  {}{0pt}
  {}
\titlespacing*{\subsection}{0pt}{0.30cm}{0.05cm}

% --- Entradas (\subsubsection): cada emprego, curso, publicação etc. --------

\titleformat{\subsubsection}
  [hang]
  {\normalsize\bfseries}
  {}{0pt}
  {}
\titlespacing*{\subsubsection}{0pt}{0.30cm}{0.04cm}

% Nada de título solto no pé da página.
\pretocmd{\section}{\Needspace*{7\baselineskip}}{}{}
\pretocmd{\subsection}{\Needspace*{5\baselineskip}}{}{}
\pretocmd{\subsubsection}{\Needspace*{4\baselineskip}}{}{}
\widowpenalty=10000
\clubpenalty=10000

% --- Data da entrada --------------------------------------------------------
% Gerada pelo filtro _cv.lua a partir de [texto]{.date} num título de entrada.

\newcommand{\cvdate}[1]{%
  \hfill{\normalfont\small\itshape\color{cvmuted}#1}%
}

% --- Listas -----------------------------------------------------------------

\setlist[itemize]{leftmargin=1.1em, topsep=0.06cm, itemsep=0.04cm, parsep=0pt}
\setlist[itemize,1]{label=\textendash}

% Rótulo das listas de data (apresentações, bolsas, prêmios) numa coluna fixa,
% senão cada item começa num recuo diferente.
\setlist[description]{
  style=multiline, leftmargin=6.4em, labelwidth=5.8em, labelsep=0.6em,
  align=left, font=\normalfont\bfseries,
  topsep=0.06cm, itemsep=0.07cm, parsep=0pt
}

% --- Tabelas (referências) --------------------------------------------------

\usepackage{booktabs}
\renewcommand{\arraystretch}{1.15}
\AtBeginEnvironment{longtable}{\small}
