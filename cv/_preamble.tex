% Formatação do currículo ("Plenário impresso"). Para editar o conteúdo,
% mexa só em cv-en.qmd e cv-pt.qmd; o README explica as convenções.
%
% A versão impressa do site: Geist no texto, Newsreader itálico em todo
% itálico do markdown (títulos, periódicos, cargos), Geist Mono nas datas,
% rótulos e números de seção. À esquerda do texto há uma coluna pendente de
% 22 mm (a coluna de datas do site) com datas, números e rótulos; o texto
% fica com uma única margem reta.
%
% Três coisas são lidas do próprio markdown, sem marcação nova:
%   - um parágrafo que é só `*Cargo*` logo depois do título da entrada é o
%     cargo, em itálico cinza na linha de baixo;
%   - um parágrafo que abre com `*Rótulo.*` (Selected delivery., Summer
%     schools and training.) sai menor e em cinza;
%   - `[[DOI]](url)` nunca fica sozinho no começo de uma linha.
%
% O contrato com o filtro _cv.lua está no cabeçalho dele: \cvnextdate,
% \cvres, \cvsep, \cvlang, \cvkicker e os ambientes cvposition, cvcontact,
% cvinterests e cvreferences.

% \cvres{DOI}: rótulo de recurso em mono, caixa-alta, no verde. O rótulo não
% se parte e não fica sozinho no começo de uma linha: o link que o contém
% gruda na palavra anterior.
\ExplSyntaxOn
\NewDocumentCommand\cvres{m}
  {
    \group_begin:
    \cvmonocaps\cvsize{7.6}{12pt}\color{cvaccent}
    \tl_set:Nn \l_tmpa_tl {#1}
    \tl_replace_all:Nnn \l_tmpa_tl { ~ } { \nobreakspace }
    \exp_args:NV \MakeUppercase \l_tmpa_tl
    \group_end:
  }
\ExplSyntaxOff
\AtBeginDocument{%
  \let\cv@linkurl\hyper@linkurl
  \def\hyper@linkurl#1#2{\cv@beforeres{#1}\cv@linkurl{#1}{#2}}}
\newcommand\cv@beforeres[1]{%
  \in@{\cvres}{#1}%
  \ifin@\ifhmode\directlua{cv_glue_resource()}\fi\fi}
% Antes de um link de recurso: se vem depois de "texto ", o espaço não
% quebra (o rótulo gruda na palavra); se vem depois de " · " (DOI · PDF),
% pode quebrar depois do ponto, nunca antes dele.
\directlua{
  local GLUE, GLYPH = node.id("glue"), node.id("glyph")
  local function nobreak_before(head, n)
    local pen = node.new("penalty")
    pen.penalty = 10000
    node.insert_before(head, n, pen)
  end
  function cv_glue_resource()
    local nest = tex.nest[tex.nest.ptr]
    local t = nest.tail
    if not t or t.id \string~= GLUE then return end
    local p = t.prev
    if p and p.id == GLYPH and p.char == 0xB7 then
      local g = p.prev
      if g and g.id == GLUE then nobreak_before(nest.head, g) end
    else
      nobreak_before(nest.head, t)
    end
  end
}

\newcommand\cv@refcell[1]{%
  \begin{tabular}[t]{@{}l@{}}#1\end{tabular}}
\ExplSyntaxOn
\seq_new:N \l__cv_refs_seq
\tl_new:N \l__cv_refrows_tl
\NewDocumentEnvironment{cvreferences}{+b}
  {
    \seq_set_split:Nnn \l__cv_refs_seq { \par } {#1}
    \seq_remove_all:Nn \l__cv_refs_seq { }
    \tl_clear:N \l__cv_refrows_tl
    \int_step_inline:nnnn { 1 } { 2 } { \seq_count:N \l__cv_refs_seq }
      {
        \tl_put_right:Ne \l__cv_refrows_tl
          {
            \exp_not:N \cv@refcell { \exp_not:e { \seq_item:Nn \l__cv_refs_seq {##1} } } &
            \exp_not:N \cv@refcell { \exp_not:e { \seq_item:Nn \l__cv_refs_seq { ##1 + 1 } } }
            \exp_not:N \\ \exp_not:N \noalign { \exp_not:N \vskip 7pt }
          }
      }
    \par
    \group_begin:
    \renewcommand\texttt[1]{{\cvmono\cvsize{8}{11.7pt}\color{cvink3}##1}}
    \renewcommand\textbf[1]{{\cvsize{9.5}{11.7pt}\bfseries\color{cvink}##1}}
    \cvsize{9}{11.7pt}\color{cvink2}
    \renewcommand\arraystretch{1}
    \setlength\tabcolsep{0pt}
    \setlength\cv@refw{\dimexpr(\linewidth-\cv@refgap)/2\relax}
    \noindent
    \begin{tabular}{@{}p{\cv@refw}@{\hspace{\cv@refgap}}p{\cv@refw}@{}}
      \tl_use:N \l__cv_refrows_tl
    \end{tabular}
    \par
    \group_end:
  }
  { }
\ExplSyntaxOff
